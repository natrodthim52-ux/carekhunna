import React, { useState } from 'react';
import { useNavigate } from 'react-router-dom';
import { motion, AnimatePresence } from 'framer-motion';
import { ChevronLeft, Plus, Pill, Trash2, Calendar, X } from 'lucide-react';
import { Input } from '@/components/ui/input';
import { Button } from '@/components/ui/button';
import { Select, SelectContent, SelectItem, SelectTrigger, SelectValue } from '@/components/ui/select';
import { Dialog, DialogContent, DialogHeader, DialogTitle } from '@/components/ui/dialog';
import { base44 } from '@/api/base44Client';
import { useQuery, useMutation, useQueryClient } from '@tanstack/react-query';
import { format } from 'date-fns';

const types = ['ยาแก้ปวด', 'ยาแก้แพ้', 'ยาแก้ไอ', 'ยาลดกรด', 'ยาธาตุ', 'ยาทาภายนอก', 'วิตามิน', 'อื่นๆ'];

const typeColors = {
  'ยาแก้ปวด': 'bg-red-50 text-red-700',
  'ยาแก้แพ้': 'bg-purple-50 text-purple-700',
  'ยาแก้ไอ': 'bg-blue-50 text-blue-700',
  'ยาลดกรด': 'bg-amber-50 text-amber-700',
  'ยาธาตุ': 'bg-green-50 text-green-700',
  'ยาทาภายนอก': 'bg-teal-50 text-teal-700',
  'วิตามิน': 'bg-orange-50 text-orange-700',
  'อื่นๆ': 'bg-gray-50 text-gray-700',
};

export default function MedicineCabinet() {
  const navigate = useNavigate();
  const queryClient = useQueryClient();
  const [showAdd, setShowAdd] = useState(false);
  const [form, setForm] = useState({ name: '', type: '', expiry_date: '', quantity: '', notes: '' });

  const { data: medicines = [] } = useQuery({
    queryKey: ['medicines'],
    queryFn: () => base44.entities.Medicine.list(),
  });

  const createMutation = useMutation({
    mutationFn: (data) => base44.entities.Medicine.create(data),
    onSuccess: () => {
      queryClient.invalidateQueries({ queryKey: ['medicines'] });
      setShowAdd(false);
      setForm({ name: '', type: '', expiry_date: '', quantity: '', notes: '' });
    },
  });

  const deleteMutation = useMutation({
    mutationFn: (id) => base44.entities.Medicine.delete(id),
    onSuccess: () => queryClient.invalidateQueries({ queryKey: ['medicines'] }),
  });

  const isExpiringSoon = (date) => {
    if (!date) return false;
    const diff = new Date(date) - new Date();
    return diff > 0 && diff < 30 * 24 * 60 * 60 * 1000;
  };

  const isExpired = (date) => {
    if (!date) return false;
    return new Date(date) < new Date();
  };

  return (
    <div className="px-5 pt-6 pb-6">
      <button onClick={() => navigate('/')} className="flex items-center gap-1 text-muted-foreground mb-4">
        <ChevronLeft className="w-5 h-5" />
        <span className="text-sm">หน้าแรก</span>
      </button>

      <motion.div initial={{ opacity: 0, y: 20 }} animate={{ opacity: 1, y: 0 }} className="space-y-5">
        <div className="flex items-center justify-between">
          <div>
            <h1 className="text-2xl font-bold text-foreground">ตู้ยาใกล้ตัว</h1>
            <p className="text-sm text-muted-foreground mt-1">ยาที่มีติดบ้าน {medicines.length} รายการ</p>
          </div>
          <Button onClick={() => setShowAdd(true)} className="rounded-2xl">
            <Plus className="w-4 h-4 mr-1" />
            เพิ่มยา
          </Button>
        </div>

        {medicines.length === 0 ? (
          <div className="text-center py-12 bg-card rounded-3xl border border-border">
            <Pill className="w-14 h-14 text-muted-foreground mx-auto mb-3" />
            <p className="text-muted-foreground font-medium">ยังไม่มียาในตู้</p>
            <p className="text-sm text-muted-foreground mt-1">กดปุ่ม "เพิ่มยา" เพื่อเริ่มบันทึก</p>
          </div>
        ) : (
          <div className="space-y-3">
            {medicines.map(med => (
              <motion.div
                key={med.id}
                layout
                className="bg-card rounded-2xl p-4 border border-border flex items-center gap-4"
              >
                <div className={`w-12 h-12 rounded-xl flex items-center justify-center shrink-0 ${typeColors[med.type] || 'bg-gray-50 text-gray-700'}`}>
                  <Pill className="w-6 h-6" />
                </div>
                <div className="flex-1 min-w-0">
                  <h3 className="font-bold text-foreground">{med.name}</h3>
                  {med.type && <span className="text-xs text-muted-foreground">{med.type}</span>}
                  <div className="flex items-center gap-2 mt-1">
                    {med.quantity && <span className="text-xs text-muted-foreground">เหลือ: {med.quantity}</span>}
                    {med.expiry_date && (
                      <span className={`text-xs flex items-center gap-1 ${isExpired(med.expiry_date) ? 'text-red-600' : isExpiringSoon(med.expiry_date) ? 'text-amber-600' : 'text-muted-foreground'}`}>
                        <Calendar className="w-3 h-3" />
                        {isExpired(med.expiry_date) ? 'หมดอายุแล้ว' : `หมดอายุ ${format(new Date(med.expiry_date), 'dd/MM/yyyy')}`}
                      </span>
                    )}
                  </div>
                </div>
                <button onClick={() => deleteMutation.mutate(med.id)} className="text-muted-foreground hover:text-destructive p-1">
                  <Trash2 className="w-4 h-4" />
                </button>
              </motion.div>
            ))}
          </div>
        )}
      </motion.div>

      {/* Add Dialog */}
      <Dialog open={showAdd} onOpenChange={setShowAdd}>
        <DialogContent className="max-w-sm mx-auto rounded-3xl">
          <DialogHeader>
            <DialogTitle>เพิ่มยาในตู้</DialogTitle>
          </DialogHeader>
          <div className="space-y-3">
            <Input placeholder="ชื่อยา" value={form.name} onChange={e => setForm({ ...form, name: e.target.value })} className="rounded-xl" />
            <Select value={form.type} onValueChange={v => setForm({ ...form, type: v })}>
              <SelectTrigger className="rounded-xl"><SelectValue placeholder="ประเภทยา" /></SelectTrigger>
              <SelectContent>
                {types.map(t => <SelectItem key={t} value={t}>{t}</SelectItem>)}
              </SelectContent>
            </Select>
            <Input type="date" placeholder="วันหมดอายุ" value={form.expiry_date} onChange={e => setForm({ ...form, expiry_date: e.target.value })} className="rounded-xl" />
            <Input placeholder="จำนวนคงเหลือ" value={form.quantity} onChange={e => setForm({ ...form, quantity: e.target.value })} className="rounded-xl" />
            <Input placeholder="หมายเหตุ (ถ้ามี)" value={form.notes} onChange={e => setForm({ ...form, notes: e.target.value })} className="rounded-xl" />
            <Button onClick={() => createMutation.mutate(form)} disabled={!form.name} className="w-full rounded-2xl">
              บันทึก
            </Button>
          </div>
        </DialogContent>
      </Dialog>
    </div>
  );
}
