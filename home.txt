import React, { useState } from 'react';
import { motion, AnimatePresence } from 'framer-motion';
import { Phone, MapPin, X, AlertTriangle } from 'lucide-react';
import { Button } from '@/components/ui/button';
import { Dialog, DialogContent, DialogHeader, DialogTitle } from '@/components/ui/dialog';
import { Input } from '@/components/ui/input';

export default function SOSButton() {
  const [open, setOpen] = useState(false);
  const [location, setLocation] = useState('');
  const [sent, setSent] = useState(false);

  const handleSOS = () => {
    setSent(true);
    setTimeout(() => {
      setSent(false);
      setOpen(false);
      setLocation('');
    }, 3000);
  };

  return (
    <>
      <motion.button
        whileHover={{ scale: 1.05 }}
        whileTap={{ scale: 0.95 }}
        onClick={() => setOpen(true)}
        className="fixed top-6 right-4 z-40 w-14 h-14 rounded-full bg-red-500 text-white shadow-lg shadow-red-500/30 flex items-center justify-center"
      >
        <AlertTriangle className="w-6 h-6" />
      </motion.button>

      <Dialog open={open} onOpenChange={setOpen}>
        <DialogContent className="bg-red-50 border-red-200 max-w-sm mx-auto">
          <DialogHeader>
            <DialogTitle className="text-red-700 text-xl flex items-center gap-2">
              <AlertTriangle className="w-6 h-6" />
              ขอความช่วยเหลือฉุกเฉิน
            </DialogTitle>
          </DialogHeader>

          {!sent ? (
            <div className="space-y-4">
              <p className="text-red-600 text-sm">
                ระบบจะส่งพิกัดและข้อมูลโรคประจำตัวของคุณไปยังหน่วยกู้ชีพทันที
              </p>

              <a
                href="tel:1669"
                className="flex items-center justify-center gap-3 w-full py-4 bg-red-600 text-white rounded-2xl text-lg font-bold shadow-lg hover:bg-red-700 transition-colors"
              >
                <Phone className="w-6 h-6" />
                โทร 1669 ทันที
              </a>

              <div className="space-y-2">
                <label className="text-sm font-medium text-red-700">
                  ระบุสถานที่เกิดเหตุ
                </label>
                <Input
                  placeholder="พิมพ์ที่อยู่หรือจุดสังเกต..."
                  value={location}
                  onChange={(e) => setLocation(e.target.value)}
                  className="border-red-300 bg-white"
                />
              </div>

              <Button
                onClick={handleSOS}
                className="w-full py-3 bg-red-700 hover:bg-red-800 text-white text-base rounded-2xl"
              >
                <MapPin className="w-5 h-5 mr-2" />
                ส่งพิกัดและข้อมูลให้ อสม.
              </Button>
            </div>
          ) : (
            <motion.div
              initial={{ scale: 0.8, opacity: 0 }}
              animate={{ scale: 1, opacity: 1 }}
              className="text-center py-6"
            >
              <div className="w-16 h-16 bg-green-500 rounded-full flex items-center justify-center mx-auto mb-3">
                <Phone className="w-8 h-8 text-white" />
              </div>
              <p className="text-lg font-bold text-green-700">ส่งข้อมูลเรียบร้อยแล้ว!</p>
              <p className="text-sm text-green-600 mt-1">อสม. จะติดต่อกลับโดยเร็วที่สุด</p>
            </motion.div>
          )}
        </DialogContent>
      </Dialog>
    </>
  );
}
