import React from 'react';
import { useNavigate } from 'react-router-dom';
import { motion } from 'framer-motion';
import { ChevronLeft, Shield, AlertTriangle, Siren, Phone, CalendarPlus, Pill, CheckCircle } from 'lucide-react';
import { Button } from '@/components/ui/button';

const triageConfig = {
  green: {
    bg: 'bg-gradient-to-b from-emerald-400 to-emerald-600',
    cardBg: 'bg-emerald-50',
    text: 'text-emerald-800',
    subText: 'text-emerald-700',
    border: 'border-emerald-200',
    icon: Shield,
    title: '🟢 อาการปกติ',
    subtitle: 'ไม่รุนแรง ดูแลตัวเองได้ที่บ้าน',
  },
  yellow: {
    bg: 'bg-gradient-to-b from-amber-400 to-amber-600',
    cardBg: 'bg-amber-50',
    text: 'text-amber-800',
    subText: 'text-amber-700',
    border: 'border-amber-200',
    icon: AlertTriangle,
    title: '🟡 ควรเฝ้าระวัง',
    subtitle: 'แนะนำให้ได้รับการตรวจเพิ่มเติม',
  },
  red: {
    bg: 'bg-gradient-to-b from-red-500 to-red-700',
    cardBg: 'bg-red-50',
    text: 'text-red-800',
    subText: 'text-red-700',
    border: 'border-red-200',
    icon: Siren,
    title: '🔴 อาการอันตราย',
    subtitle: 'ต้องพบแพทย์ทันที!',
  },
};

export default function TriageResult({ result, onBack }) {
  const navigate = useNavigate();
  const level = result.triage_level || 'green';
  const config = triageConfig[level];
  const Icon = config.icon;

  return (
    <div className={`min-h-screen ${config.bg} px-5 pt-6 pb-24`}>
      <button onClick={onBack} className="flex items-center gap-1 text-white/80 mb-4">
        <ChevronLeft className="w-5 h-5" />
        <span className="text-sm">คัดกรองใหม่</span>
      </button>

      <motion.div
        initial={{ opacity: 0, y: 30 }}
        animate={{ opacity: 1, y: 0 }}
        className="space-y-5"
      >
        {/* Header */}
        <div className="text-center text-white py-4">
          <motion.div
            initial={{ scale: 0 }}
            animate={{ scale: 1 }}
            transition={{ type: 'spring', delay: 0.2 }}
            className="w-20 h-20 bg-white/20 rounded-full flex items-center justify-center mx-auto mb-3"
          >
            <Icon className="w-10 h-10" />
          </motion.div>
          <h1 className="text-2xl font-bold">{config.title}</h1>
          <p className="text-white/90 mt-1">{config.subtitle}</p>
        </div>

        {/* Condition */}
        <div className={`${config.cardBg} ${config.border} border rounded-3xl p-5`}>
          <h2 className={`text-lg font-bold ${config.text}`}>
            {result.condition_name}
          </h2>
          <p className={`text-sm ${config.subText} mt-2 leading-relaxed`}>
            {result.description}
          </p>
        </div>

        {/* Self Care */}
        {result.self_care?.length > 0 && (
          <div className={`${config.cardBg} ${config.border} border rounded-3xl p-5`}>
            <h3 className={`font-bold ${config.text} mb-3`}>💊 วิธีดูแลตัวเอง</h3>
            <ul className="space-y-2">
              {result.self_care.map((tip, i) => (
                <li key={i} className={`flex items-start gap-2 text-sm ${config.subText}`}>
                  <CheckCircle className="w-4 h-4 mt-0.5 shrink-0" />
                  <span>{tip}</span>
                </li>
              ))}
            </ul>
          </div>
        )}

        {/* Recommended Medicines */}
        {result.recommended_medicines?.length > 0 && (
          <div className={`${config.cardBg} ${config.border} border rounded-3xl p-5`}>
            <h3 className={`font-bold ${config.text} mb-3`}>
              <Pill className="w-4 h-4 inline mr-1" />
              ยาที่แนะนำ
            </h3>
            <div className="flex flex-wrap gap-2">
              {result.recommended_medicines.map((med, i) => (
                <span key={i} className={`px-3 py-1 ${config.border} border rounded-full text-sm ${config.subText}`}>
                  {med}
                </span>
              ))}
            </div>
            <Button
              variant="outline"
              className="mt-3 w-full rounded-2xl"
              onClick={() => navigate('/medicine')}
            >
              <Pill className="w-4 h-4 mr-2" />
              เช็กยาในตู้ยาของคุณ
            </Button>
          </div>
        )}

        {/* Warning Signs */}
        {result.warning_signs?.length > 0 && (
          <div className={`${config.cardBg} ${config.border} border rounded-3xl p-5`}>
            <h3 className={`font-bold ${config.text} mb-3`}>⚠️ อาการที่ต้องเฝ้าระวัง</h3>
            <ul className="space-y-2">
              {result.warning_signs.map((sign, i) => (
                <li key={i} className={`flex items-start gap-2 text-sm ${config.subText}`}>
                  <AlertTriangle className="w-4 h-4 mt-0.5 shrink-0" />
                  <span>{sign}</span>
                </li>
              ))}
            </ul>
          </div>
        )}

        {/* Actions based on level */}
        {level === 'yellow' && (
          <Button
            className="w-full h-14 rounded-2xl text-base font-bold bg-amber-800 hover:bg-amber-900"
            onClick={() => navigate('/volunteers')}
          >
            <CalendarPlus className="w-5 h-5 mr-2" />
            นัดหมาย อสม. ตรวจเพิ่มเติม
          </Button>
        )}

        {level === 'red' && (
          <div className="space-y-3">
            <a
              href="tel:1669"
              className="flex items-center justify-center gap-3 w-full py-4 bg-white text-red-600 rounded-2xl text-lg font-bold shadow-lg"
            >
              <Phone className="w-6 h-6" />
              โทร 1669 ทันที
            </a>
            <Button
              variant="outline"
              className="w-full rounded-2xl border-white/30 text-white hover:bg-white/10"
              onClick={() => navigate('/volunteers')}
            >
              ติดต่อ อสม. ในพื้นที่
            </Button>
          </div>
        )}

        {/* Back to home */}
        <Button
          variant="ghost"
          className="w-full text-white/80 hover:text-white hover:bg-white/10 rounded-2xl"
          onClick={() => navigate('/')}
        >
          กลับหน้าแรก
        </Button>
      </motion.div>
    </div>
  );
}
